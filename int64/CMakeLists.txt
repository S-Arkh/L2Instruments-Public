cmake_minimum_required(VERSION 3.24)
project(UccInt64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_executable(int64_semantics_test tests/int64_semantics_test.cpp)
target_include_directories(int64_semantics_test PRIVATE src)
add_test(NAME int64_semantics COMMAND int64_semantics_test)

if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR NOT (MSVC OR MINGW))
        message(FATAL_ERROR "Use Visual Studio 2022 -A Win32 or MinGW-w64 i686")
    endif()
    # MinHook declares an older cmake_minimum_required; keep the static CRT
    # selection effective in that subproject as well.
    set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
    # Keep the complete hook dependency with the sources for offline rebuilds.
    add_subdirectory(third_party/minhook)
    add_library(Ucc64Bridge SHARED src/editor_bridge.cpp)
    target_link_libraries(Ucc64Bridge PRIVATE minhook)
    add_executable(ucc64 src/launcher.cpp)
    target_link_libraries(ucc64 PRIVATE bcrypt psapi)
    add_dependencies(ucc64 Ucc64Bridge)
    add_executable(client_native_probe tests/client_native_probe.cpp)
    target_include_directories(client_native_probe PRIVATE src)
    if(MSVC)
        target_sources(Ucc64Bridge PRIVATE src/ucc64.def)
        foreach(target Ucc64Bridge ucc64 client_native_probe int64_semantics_test)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
        endforeach()
    else()
        set_target_properties(Ucc64Bridge PROPERTIES PREFIX "")
        target_link_options(Ucc64Bridge PRIVATE -static -Wl,--kill-at)
        foreach(target ucc64 client_native_probe)
            target_link_options(${target} PRIVATE -static -municode)
        endforeach()
        target_link_options(int64_semantics_test PRIVATE -static)
    endif()
endif()
