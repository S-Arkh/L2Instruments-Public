cmake_minimum_required(VERSION 3.20)
project(InterfaceEditor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${PYBIND_DIR}")
find_package(pybind11 CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import nlohmann_json; print(nlohmann_json.get_include())" OUTPUT_VARIABLE JSON_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
add_library(interface_core STATIC core/document.cpp core/textures.cpp core/dat.cpp)
target_link_libraries(interface_core PUBLIC ${CMAKE_DL_LIBS})
target_include_directories(interface_core PUBLIC core "${JSON_INCLUDE}")
if(MSVC)
    target_compile_options(interface_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(interface_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
pybind11_add_module(_interface_core bindings/module.cpp)
target_link_libraries(_interface_core PRIVATE interface_core)
set_target_properties(_interface_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/app")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(_interface_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/app" RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/app")
endforeach()
