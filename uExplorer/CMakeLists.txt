cmake_minimum_required(VERSION 3.24)
project(uExplorer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(uexplorer_core SHARED core/package.cpp core/decompiler.cpp core/metadata.cpp core/resources.cpp core/volumes.cpp core/materials.cpp core/animated_resources.cpp core/api.cpp)
find_package(Python COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import nlohmann_json; print(nlohmann_json.get_include())"
  OUTPUT_VARIABLE JSON_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
target_include_directories(uexplorer_core PRIVATE "${JSON_INCLUDE}")
target_compile_definitions(uexplorer_core PRIVATE UEX_BUILD)
if(MSVC)
  target_compile_options(uexplorer_core PRIVATE /W4 /EHsc /utf-8 /bigobj)
else()
  target_compile_options(uexplorer_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(uexplorer_cli core/cli.cpp)
target_link_libraries(uexplorer_cli PRIVATE uexplorer_core)
install(TARGETS uexplorer_core uexplorer_cli RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
